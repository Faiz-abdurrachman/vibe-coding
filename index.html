LANGKAH 4 • MENIT 45–75

Supabase DB & Defense in Depth

SQL Supabase (Salin & Run) 
CREATE TABLE public.cash_transactions (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    tanggal DATE NOT NULL DEFAULT CURRENT_DATE,
    jenis_transaksi TEXT NOT NULL CHECK (jenis_transaksi IN ('Masuk', 'Keluar')),
    nominal NUMERIC(12, 2) NOT NULL CHECK (nominal > 0), -- Kunci TC-01 di Database!
    keterangan TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE public.cash_transactions ENABLE ROW LEVEL SECURITY;

CREATE POLICY "CRUD Publik" ON public.cash_transactions 
FOR ALL TO anon USING (true) WITH CHECK (nominal > 0);
CHECK (nominal > 0) mengunci aturan di level engine PostgreSQL.
